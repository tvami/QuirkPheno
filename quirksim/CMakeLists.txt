file(GLOB QSIM_SRC ${CMAKE_CURRENT_SOURCE_DIR}/src/*.cc)
add_library(qsimlib STATIC ${QSIM_SRC})
target_include_directories(qsimlib PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(qsimlib PUBLIC quirkcore)
if(QSIM_WITH_HEPMC3)
  target_compile_definitions(qsimlib PUBLIC QSIM_WITH_HEPMC3)
  target_include_directories(qsimlib SYSTEM PUBLIC ${HEPMC3_INCLUDE_DIR})
  target_link_libraries(qsimlib PUBLIC ${HEPMC3_LIBRARIES})
endif()
if(QSIM_WITH_PYTHIA8)
  target_compile_definitions(qsimlib PUBLIC QSIM_WITH_PYTHIA8)
  target_include_directories(qsimlib SYSTEM PUBLIC ${PYTHIA8_INC})
  target_link_libraries(qsimlib PUBLIC ${PYTHIA8_LIB})
endif()

add_executable(quirksim quirksim.cc)
target_link_libraries(quirksim qsimlib)
if(QSIM_WITH_PYTHIA8 AND QSIM_WITH_HEPMC3)
  add_executable(quirkgen quirkgen.cc)
  target_link_libraries(quirkgen qsimlib)
endif()
install(TARGETS quirksim quirkcore)
