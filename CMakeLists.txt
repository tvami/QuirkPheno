cmake_minimum_required(VERSION 3.16)
project(QuirkStandalone CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

# geant4-config, not Geant4Config.cmake: the latter needs X11/Motif/OpenGL dev packages absent in el9 images
find_program(GEANT4_CONFIG geant4-config REQUIRED)
execute_process(COMMAND ${GEANT4_CONFIG} --cflags OUTPUT_VARIABLE G4_CFLAGS OUTPUT_STRIP_TRAILING_WHITESPACE)
execute_process(COMMAND ${GEANT4_CONFIG} --libs OUTPUT_VARIABLE G4_LIBS OUTPUT_STRIP_TRAILING_WHITESPACE)
execute_process(COMMAND ${GEANT4_CONFIG} --version OUTPUT_VARIABLE G4_VERSION OUTPUT_STRIP_TRAILING_WHITESPACE)
separate_arguments(G4_CFLAGS UNIX_COMMAND "${G4_CFLAGS}")
separate_arguments(G4_LIBS UNIX_COMMAND "${G4_LIBS}")
list(FILTER G4_CFLAGS EXCLUDE REGEX "^-flto|^-std=|^-O")
list(FILTER G4_LIBS EXCLUDE REGEX "^-flto|OpenGL|ToolsSG|Qt3D|Vtk|G4vis|G4FR|G4RayTracer|G4VRML|G4Tree|G4GMocren|G4visHepRep|G4visXXX|G4modeling|G4interfaces|G4gdml|G4geomtext|G3toG4")
message(STATUS "Geant4 ${G4_VERSION} from ${GEANT4_CONFIG}")
option(QSIM_WITH_HEPMC3 "HepMC3 input" ON)
option(QSIM_WITH_PYTHIA8 "built-in Pythia8 and quirkgen" ON)
if(QSIM_WITH_HEPMC3)
  find_package(HepMC3 REQUIRED)
endif()
if(QSIM_WITH_PYTHIA8)
  # pythia8-config, as in LCG
  find_program(PYTHIA8_CONFIG pythia8-config REQUIRED)
  execute_process(COMMAND ${PYTHIA8_CONFIG} --includedir OUTPUT_VARIABLE PYTHIA8_INC OUTPUT_STRIP_TRAILING_WHITESPACE)
  execute_process(COMMAND ${PYTHIA8_CONFIG} --libdir OUTPUT_VARIABLE PYTHIA8_LIBDIR OUTPUT_STRIP_TRAILING_WHITESPACE)
  find_library(PYTHIA8_LIB pythia8 PATHS ${PYTHIA8_LIBDIR} NO_DEFAULT_PATH REQUIRED)
endif()

add_subdirectory(quirkcore)
add_subdirectory(quirksim)
